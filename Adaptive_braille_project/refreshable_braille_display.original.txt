/*

The objective of this program is to take a string and represent it
using the braille display one letter at a time.

The braille alphabet uses characters with six dots
in three rows and two columns. the dots are labeled 1-6
as follows:

  1  4
  2  5
  3  6

*/
#include <Arduino.h>
#include <HardwareSerial.h>
#include <DFRobotDFPlayerMini.h>
#include <LittleFS.h>

// Button 1 = Random vocabulary word
#define WORD_BTN 12

// Button 2 = Next letter
#define SPELL_BTN 13

// ============================================================
// DFPLAYER MINI
// ============================================================

// ESP32 RX2 <- DFPlayer TX
// ESP32 TX2 -> DFPlayer RX
#define DF_RX_PIN 16
#define DF_TX_PIN 17

HardwareSerial dfSerial(2);
DFRobotDFPlayerMini dfPlayer;

// store the code for each letter a-z in a 2D array
int braille[26][6] = {
  { 1, 0, 0, 0, 0, 0 },  // a (first bump is raised, others are lowered)
  { 1, 1, 0, 0, 0, 0 },  // b (first and second bumps are raised)
  { 1, 0, 0, 1, 0, 0 },  // c (and so on...)
  { 1, 0, 0, 1, 1, 0 },  // d
  { 1, 0, 0, 0, 1, 0 },  // e
  { 1, 1, 0, 1, 0, 0 },  // f
  { 1, 1, 0, 1, 1, 0 },  // g
  { 1, 1, 0, 0, 1, 0 },  // h
  { 0, 1, 0, 1, 0, 0 },  // i
  { 0, 1, 0, 1, 1, 0 },  // j
  { 1, 0, 1, 0, 0, 0 },  // k
  { 1, 1, 1, 0, 0, 0 },  // l
  { 1, 0, 1, 1, 0, 0 },  // m
  { 1, 0, 1, 1, 1, 0 },  // n
  { 1, 0, 1, 0, 1, 0 },  // o
  { 1, 1, 1, 1, 0, 0 },  // p
  { 1, 1, 1, 1, 1, 0 },  // q
  { 1, 1, 1, 0, 1, 0 },  // r
  { 0, 1, 1, 1, 0, 0 },  // s
  { 0, 1, 1, 1, 1, 0 },  // t
  { 1, 0, 1, 0, 0, 1 },  // u
  { 1, 1, 1, 0, 0, 1 },  // v
  { 0, 1, 0, 1, 1, 1 },  // w
  { 1, 0, 1, 1, 0, 1 },  // x
  { 1, 0, 1, 1, 1, 1 },  // y
  { 1, 0, 1, 0, 1, 1 },  // z
};

// array of the ESP32 pins used to control the solenoids
int BRAILLE_PINS[6] = { 32, 33, 25, 26, 27, 14 };


const int MAX_WORDS = 255;

struct WordEntry {
  String word;
  int audioNum;
};

int wordCount = 0;
bool endOfWord = false;
WordEntry wordList[MAX_WORDS];
String currentWord = "";

int currentAudio = 0;


char letter[] = "abcdefghijklmnopqrstuvwxyz";
int letterIndex ;







void setup() {  // setup code that only runs once
  // initialize serial communication (allows printing to serial monitor)
  Serial.begin(115200);
  delay(1000);
  Serial.println();
  Serial.println("========================================");
  Serial.println("   BRAILLE WORD LEARNING DEVICE");
  Serial.println("========================================");

  
  // set control pins as outputs
  for (int i = 0; i <= 5; i++) {
    pinMode(BRAILLE_PINS[i], OUTPUT);
    digitalWrite(BRAILLE_PINS[i], LOW);
  }

  Serial.println("[OK] Braille outputs ready.");

  // ==========================================================
  // BUTTONS
  // ==========================================================

  Serial.println();
  Serial.println("[2] Initializing buttons...");

  pinMode(WORD_BTN, INPUT_PULLUP);
  pinMode(SPELL_BTN, INPUT_PULLUP);

  Serial.println("[OK] Buttons ready.");

    // ==========================================================
  // LITTLEFS
  // ==========================================================
  // false = do NOT automatically format filesystem
  if (!LittleFS.begin(false)) 
    Serial.println("[ERROR] LittleFS mount failed!");

  else
    Serial.println("[OK]LittleFS mounted");

  Serial.print("LittleFS total bytes: ");
  Serial.println(LittleFS.totalBytes());

  Serial.print("LittleFS used bytes: ");
  Serial.println(LittleFS.usedBytes());

  loadWordList();

    // ==========================================================
  // DFPLAYER
  // ==========================================================

  Serial.println();
  Serial.println("[4] Starting DFPlayer Mini...");

  dfSerial.begin(
    9600,
    SERIAL_8N1,
    DF_RX_PIN,
    DF_TX_PIN
  );

  delay(1000);

  if (!dfPlayer.begin(dfSerial, true, true)) {

    Serial.println("[ERROR] DFPlayer Mini not found!");

    Serial.println();
    Serial.println("Check:");
    Serial.println("  - DFPlayer VCC");
    Serial.println("  - DFPlayer GND");
    Serial.println("  - ESP32 RX/TX wiring");
    Serial.println("  - 1k resistor on DFPlayer RX");
    Serial.println("  - SD card");
    Serial.println("  - Audio files");

    while (true) {
      delay(1000);
    }
  }

  dfPlayer.volume(20);

  Serial.println("[OK] DFPlayer Mini ready.");


  // ==========================================================
  // RANDOM NUMBER GENERATOR
  // ==========================================================

  randomSeed(esp_random());


  // ==========================================================
  // READY
  // ==========================================================

  Serial.println();
  Serial.println("========================================");
  Serial.println("[READY]");
  Serial.println();
  Serial.println("Button 1 -> Random word");
  Serial.println("Button 2 -> Next letter");
  Serial.println();
  Serial.println("Words loaded:");
  Serial.println(wordCount);
  Serial.println("========================================");  

}

void loop() {  // code that loops forever
  Serial.print("press buttons\n");
  while (digitalRead(WORD_BTN) == HIGH) 
    {
         
    }
      // wait for button release
  while (digitalRead(WORD_BTN) == LOW) 
    {
      
    }
  delay(1000);
  selectRandomWord();  // random word select and play
  delay(500);
  
  for (int j=0; j < currentWord.length(); j++)
  {
    while (digitalRead(SPELL_BTN) == HIGH) 
    {
         
    }
      // wait for button release
    while (digitalRead(SPELL_BTN) == LOW) 
    {
      
    }
    Serial.print(j);
    Serial.print("\n");
    spellNextLetter(currentWord[j]);
    Serial.print("current letter\n");
    Serial.print(currentWord[j]);
    BrailleShow();
    delay(500);
  }
  
  clearBraille();
  endOfWord = false;
}



void loadWordList() {

  Serial.println();
  Serial.println("========================================");
  Serial.println(" Loading words.txt from LittleFS");
  Serial.println("========================================");

   // ----------------------------------------------------------
  // Check if file exists
  // ----------------------------------------------------------

  if (!LittleFS.exists("/words.txt")) {

    Serial.println("[ERROR] /words.txt does not exist!");
    return;
  }
  else
    Serial.println("Files currently in LittleFS:");

  

  File file = LittleFS.open("/words.txt", "r");

  if (!file) {

    Serial.println("[ERROR] Could not open /words.txt!");
    return;
  }

  //----------------------------------------------------------
  // Show file size
  // ----------------------------------------------------------

  Serial.print("[OK] /words.txt found.");

  Serial.print(" Size = ");

  Serial.print(file.size());

  Serial.println(" bytes");


  if (file.size() == 0) {

    Serial.println("[ERROR] /words.txt is empty!");

    file.close();

    return;
  }
 
  
    // ----------------------------------------------------------
  // Read words
  // ----------------------------------------------------------

  wordCount = 0;

  while (
    file.available() &&
    wordCount < MAX_WORDS
  ) {

    String word = file.readStringUntil('\n');

    word.trim();


    // Ignore empty lines
    if (word.length() == 0) {
      continue;
    }


    // Convert to lowercase
    word.toLowerCase();


    // Save word
    wordList[wordCount].word = word;


    // Audio numbering
    //
    // wordCount = 0 -> 27
    // wordCount = 1 -> 28
    // wordCount = 2 -> 29
    //

    wordList[wordCount].audioNum = wordCount + 1;


    wordCount++;
  }


  file.close();


  // ----------------------------------------------------------
  // Print result
  // ----------------------------------------------------------

  Serial.println();

  Serial.print("[OK] Loaded ");

  Serial.print(wordCount);

  Serial.println(" vocabulary words successfully!");


  // ----------------------------------------------------------
  // Display first 10 words
  // ----------------------------------------------------------

  Serial.println();

  Serial.println("First 10 words:");

  Serial.println("----------------------------------------");

  for (
    int i = 0;
    i < 20 && i < wordCount;
    i++
  ) {

    Serial.print(i + 1);

    Serial.print(": ");

    Serial.print(wordList[i].word);

    Serial.print(" -> ");

    Serial.printf(
      "%d.mp3",
      wordList[i].audioNum
    );

    Serial.println();
  }

  Serial.println("----------------------------------------");


  // ----------------------------------------------------------
  // Display last word
  // ----------------------------------------------------------

  if (wordCount > 0) {

    Serial.print("Last word: ");

    Serial.print(wordList[wordCount - 1].word);

    Serial.print(" -> ");

    Serial.printf(
      "%d.mp3",
      wordList[wordCount - 1].audioNum
    );

    Serial.println();
  }

  Serial.println("========================================");
}


void selectRandomWord() {

  // ----------------------------------------------------------
  // Check word list
  // ----------------------------------------------------------

  if (wordCount == 0) {

    Serial.println();

    Serial.println("[WARN] Word list is empty.");

    Serial.println("Check /words.txt.");

    return;
  }
  // ----------------------------------------------------------
  // Select random word
  // ----------------------------------------------------------

  int index = random(wordCount);

  currentWord = wordList[index].word;

  currentAudio = wordList[index].audioNum;
  dfPlayer.playFolder(2, currentAudio);
  Serial.print("word\n");
  Serial.print(currentWord);
  Serial.print("\n");
  Serial.print("audio\n");
  Serial.print(currentAudio);
  Serial.print("\n");
    
  endOfWord = true;

}

void spellNextLetter(char alphabet) {

  // ----------------------------------------------------------
  // No word selected
  // ----------------------------------------------------------

  if (currentWord.length() == 0) {

    Serial.println();

    Serial.println("[WARN] No word selected.");

    Serial.println("Press Button 1 first.");

    return;
  }
  // convert uppercase to lowercase
  if (alphabet >= 'A' && alphabet <= 'Z') 
  {
    alphabet = alphabet + ('a' - 'A');
  }


  // calculating index

  for (int j = 0; j < 26; j++)
  {
    if (alphabet == letter[j]) 
    {
      letterIndex = j;
      break;
    }
  }

  if(endOfWord)
  {
    dfPlayer.playFolder(1, (letterIndex + 1));
  }

}

void BrailleShow()
{
  for (int k = 0; k <= 5; k++) 
  {
    digitalWrite(BRAILLE_PINS[k], braille[letterIndex][k]);
    Serial.print(braille[letterIndex][k]);
    Serial.print("  ");
  }

}
void clearBraille() {

  for (int i = 0; i <= 5; i++) {

    digitalWrite(BRAILLE_PINS[i],LOW);
  }
}
